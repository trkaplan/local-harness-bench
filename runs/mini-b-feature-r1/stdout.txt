class JsonStore:
    def __init__(self, path: Path):
        self.path = Path(path)

    def load(self) -> list[Task]:
        if not self.path.exists():
            return []
        data = json.loads(self.path.read_text() or "[]")
        return [Task.from_dict(d) for d in data]

    def save(self, tasks: list[Task]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(json.dumps([t.to_dict() for t in tasks], indent=2))

    def next_id(self, tasks: list[Task]) -> int:
        return (max((t.id for t in tasks), default=0) + 1)

from .models import Task
from .storage import JsonStore


class TodoService:
    def __init__(self, store: JsonStore):
        self.store = store

    def add(self, title: str, tags: list[str] | None = None) -> Task:
        tasks = self.store.load()
        t = Task.new(self.store.next_id(tasks), title, tags)
        tasks.append(t)
        self.store.save(tasks)
        return t

    def list(self, include_done: bool = False, tag: str | None = None) -> 
list[Task]:
        tasks = self.store.load()
        out = [t for t in tasks if include_done or not t.done]
        if tag:
            out = [t for t in out if tag in (t.tags or [])]
        return out

    def complete(self, task_id: int) -> Task:
        tasks = self.store.load()
        for t in tasks:
            if t.id == task_id:
                t.done = True
                self.store.save(tasks)
                return t
        raise KeyError(task_id)

    def remove(self, task_id: int) -> None:
        tasks = self.store.load()
        new = [t for t in tasks if t.id != task_id]
        if len(new) == len(tasks):
            raise KeyError(task_id)
        self.store.save(new)

```
"""SQLite storage backend using the standard library."""
from __future__ import annotations
import json
import sqlite3
from pathlib import Path
from .models import Task


class SqliteStore:
    def __init__(self, path: Path):
        self.path = Path(path)

    def _connect(self) -> sqlite3.Connection:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        conn = sqlite3.connect(self.path)
        conn.execute(
            "CREATE TABLE IF NOT EXISTS tasks ("
            "id INTEGER PRIMARY KEY, "
            "title TEXT NOT NULL, "
            "done INTEGER NOT NULL DEFAULT 0, "
            "created_at TEXT NOT NULL, "
            "tags TEXT NOT NULL)"
        )
        return conn

    def load(self) -> list[Task]:
        if not self.path.exists():
            return []
        conn = self._connect()
        try:
            rows = conn.execute("SELECT id, title, done, created_at, tags FROM 
tasks ORDER BY id").fetchall()
        finally:
            conn.close()
        return [Task(id=r[0], title=r[1], done=bool(r[2]), created_at=r[3], 
tags=json.loads(r[4])) for r in rows]

    def save(self, tasks: list[Task]) -> None:
        conn = self._connect()
        try:
            with conn:
                conn.execute("DELETE FROM tasks")
                conn.executemany(
                    "INSERT INTO tasks (id, title, done, created_at, tags) 
VALUES (?, ?, ?, ?, ?)",
                    [(t.id, t.title, int(t.done), t.created_at, 
json.dumps(t.tags or [])) for t in tasks],
                )
        finally:
            conn.close()

    def next_id(self, tasks: list[Task]) -> int:
        return (max((t.id for t in tasks), default=0) + 1)
```

```
# todo

A small CLI todo manager. The default storage backend is SQLite (`~/.todo.db`);
JSON files are still supported — any `--file` path ending in `.json` uses the
JSON backend. Override the default path with `--file` or the `TODO_FILE` env
var.

Examples:

```
python -m todo.cli add "write tests" --tag dev
python -m todo.cli rename 1 "rewrite tests"
python -m todo.cli migrate old.json new.db   # copies tasks between backends
