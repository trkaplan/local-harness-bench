    def add(self, title: str, tags: list[str] | None = None) -> Task:
        tasks = self.store.load()
        t = Task.new(self.store.next_id(tasks), title, tags)
        tasks.append(t)
        self.store.save(tasks)
        return t

    def list(self, include_done: bool = False, tag: str | None = None) -> 
list[Task]:
        tasks = self.store.load()
        out = [t for t in tasks if include_done or not t.done]
        if tag:
            out = [t for t in out if tag in (t.tags or [])]
        return out

    def complete(self, task_id: int) -> Task:
        tasks = self.store.load()
        for t in tasks:
            if t.id == task_id:
                t.done = True
                self.store.save(tasks)
                return t
        raise KeyError(task_id)

    def remove(self, task_id: int) -> None:
        tasks = self.store.load()
        new = [t for t in tasks if t.id != task_id]
        if len(new) == len(tasks):
            raise KeyError(task_id)

"""SQLite storage backend using the standard library."""
from __future__ import annotations
import json
import sqlite3
from pathlib import Path
from .models import Task


class SqliteStore:
    def __init__(self, path: Path):
        self.path = Path(path)

    def _connect(self) -> sqlite3.Connection:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        conn = sqlite3.connect(str(self.path))
        conn.execute(
            """CREATE TABLE IF NOT EXISTS tasks (
                   id INTEGER PRIMARY KEY,
                   title TEXT NOT NULL,
                   done INTEGER NOT NULL DEFAULT 0,
                   created_at TEXT NOT NULL,
                   tags TEXT NOT NULL
               )"""
        )
        return conn

    def load(self) -> list[Task]:
        if not self.path.exists():
            return []
        conn = self._connect()
        try:
            rows = conn.execute(
                "SELECT id, title, done, created_at, tags FROM tasks ORDER BY 
id"
            ).fetchall()
        finally:
            conn.close()
        return [
            Task(id=r[0], title=r[1], done=bool(r[2]), created_at=r[3], 
tags=json.loads(r[4]))
            for r in rows
        ]

    def save(self, tasks: list[Task]) -> None:
        conn = self._connect()
        try:
            with conn:
                conn.execute("DELETE FROM tasks")
                conn.executemany(
                    "INSERT INTO tasks (id, title, done, created_at, tags) 
VALUES (?, ?, ?, ?, ?)",
                    [
                        (t.id, t.title, 1 if t.done else 0, t.created_at, 
json.dumps(list(t.tags or [])))
                        for t in tasks
                    ],
                )
        finally:
            conn.close()

    def next_id(self, tasks: list[Task]) -> int:
        return (max((t.id for t in tasks), default=0) + 1)

"""Storage backends. The whole task list is rewritten on every change."""
from __future__ import annotations
import json
from pathlib import Path
from .models import Task
from .sqlite_store import SqliteStore


class JsonStore:
    def __init__(self, path: Path):
        self.path = Path(path)

    def load(self) -> list[Task]:
        if not self.path.exists():
            return []
        data = json.loads(self.path.read_text() or "[]")
        return [Task.from_dict(d) for d in data]

    def save(self, tasks: list[Task]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(json.dumps([t.to_dict() for t in tasks], indent=2))

    def next_id(self, tasks: list[Task]) -> int:
        return (max((t.id for t in tasks), default=0) + 1)


def open_store(path: str | Path):
    """Return the store backend matching the file extension."""
    path = Path(path)
    if path.suffix == ".json":
        return JsonStore(path)
    return SqliteStore(path)

# todo

A small CLI todo manager. By default tasks are stored in a SQLite database at
`~/.todo.db`; the JSON backend is still available and is used automatically when
the storage path ends with `.json`. The `--file` option (or the `TODO_FILE`
environment variable) selects the storage path.

```bash
python -m todo.cli add "x"                 # uses ~/.todo.db
python -m todo.cli --file t.json add "x"   # JSON backend
python -m todo.cli rename 1 "new title"    # rename task #1
```

## Migrating between backends

The `migrate` subcommand copies all tasks from one store to another (either
direction, chosen by file extension), preserving ids, done flags, created_at and
tags. The destination must be empty:

```bash
python -m todo.cli migrate ~/.todo.json ~/.todo.db   # migrated N tasks
```
