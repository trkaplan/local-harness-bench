"""SQLite storage backend using the standard library sqlite3 module."""
from __future__ import annotations
import json
import sqlite3
from pathlib import Path
from .models import Task


class SqliteStore:
    def __init__(self, path: Path):
        self.path = Path(path)

    def _connect(self) -> sqlite3.Connection:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        conn = sqlite3.connect(self.path)
        conn.execute(
            """CREATE TABLE IF NOT EXISTS tasks (
                   id INTEGER PRIMARY KEY,
                   title TEXT NOT NULL,
                   done INTEGER NOT NULL DEFAULT 0,
                   created_at TEXT NOT NULL,
                   tags TEXT NOT NULL
               )"""
        )
        return conn

    def load(self) -> list[Task]:
        if not self.path.exists():
            return []
        conn = self._connect()
        try:
            rows = conn.execute(
                "SELECT id, title, done, created_at, tags FROM tasks ORDER BY 
id"
            ).fetchall()
        finally:
            conn.close()
        return [
            Task(id=r[0], title=r[1], done=bool(r[2]), created_at=r[3], 
tags=json.loads(r[4]))
            for r in rows
        ]

    def save(self, tasks: list[Task]) -> None:
        conn = self._connect()
        try:
            with conn:
                conn.execute("DELETE FROM tasks")
                conn.executemany(
                    "INSERT INTO tasks (id, title, done, created_at, tags) 
VALUES (?, ?, ?, ?, ?)",
                    [
                        (t.id, t.title, 1 if t.done else 0, t.created_at, 
json.dumps(list(t.tags or [])))
                        for t in tasks
                    ],
                )
        finally:
            conn.close()

    def next_id(self, tasks: list[Task]) -> int:
        return (max((t.id for t in tasks), default=0) + 1)

"""Storage backends. The whole task list is rewritten on every change."""
from __future__ import annotations
import json
from pathlib import Path
from .models import Task
from .sqlite_store import SqliteStore


class JsonStore:
    def __init__(self, path: Path):
        self.path = Path(path)

    def load(self) -> list[Task]:
        if not self.path.exists():
            return []
        data = json.loads(self.path.read_text() or "[]")
        return [Task.from_dict(d) for d in data]

    def save(self, tasks: list[Task]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(json.dumps([t.to_dict() for t in tasks], indent=2))

    def next_id(self, tasks: list[Task]) -> int:
        return (max((t.id for t in tasks), default=0) + 1)


def open_store(path: str | Path):
    """Return the storage backend matching the file extension.

    ``.json`` paths use JsonStore; anything else uses SqliteStore.
    """
    path = Path(path)
    if path.suffix == ".json":
        return JsonStore(path)
    return SqliteStore(path)

# todo

A small CLI todo manager. The default storage backend is a SQLite database at
`~/.todo.db` (override with `--file PATH` or the `TODO_FILE` env var). A JSON
backend is still available: any `--file` path ending in `.json` uses it.

```bash
todo add "write tests" --tag dev      # uses ~/.todo.db by default
todo list                             # --all to include done tasks
todo done 1
todo rename 1 "better title"
todo rm 1
todo --file t.json add "json-backed"  # JSON backend via extension
```

The `migrate` subcommand copies all tasks between stores (either direction,
chosen by file extension), preserving ids, done flags, created_at and tags:

```bash
todo migrate ~/.todo.json ~/.todo.db   # JSON -> SQLite, prints "migrated N 
tasks"
todo migrate old.db new.json           # SQLite -> JSON
```

Migrating into a non-empty destination fails with exit code 1 and leaves the
destination untouched.

Tests: `python -m pytest -q`.
